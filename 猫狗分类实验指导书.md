## 0. 实验环境与公共代码

### 0.1 理论要点

| 概念 | |
|---|---|
| 卷积层（Conv2d） | 用一个小窗口（如 3×3）在图片上滑动扫描局部特征（边缘、纹理、耳朵形状……） |
| 池化层（MaxPool2d） | 把特征图缩小一半，保留最显著的响应，同时扩大感受野 |
| BatchNorm | 把每层输入拉回稳定分布，让训练更快更稳 |
| 残差连接（ResNet 的"+x"） | 让梯度可以"抄近道"传回浅层，网络再深也训得动 |
| 全连接层（Linear） | 把卷积提取的特征图"汇总投票"，输出每个类别的得分 |

### 0.2 公共代码（数据加载，每个实验都会用到）

保存为 `common.py`，后面每一步直接 `from common import ...`。

```python
import os
from PIL import Image
import torch
from torch.utils.data import Dataset, DataLoader, random_split
from torchvision import transforms

IMAGENET_MEAN = [0.485, 0.456, 0.406]
IMAGENET_STD  = [0.229, 0.224, 0.225]

class FileNameDataset(Dataset):
    """从文件名解析标签：cat.x.jpg -> 0，dog.x.jpg -> 1"""
    def __init__(self, image_dir, transform=None):
        self.image_dir = image_dir
        self.transform = transform
        self.images = [f for f in os.listdir(image_dir)
                       if f.lower().endswith(('.jpg', '.jpeg', '.png'))]

    def __len__(self):
        return len(self.images)

    def __getitem__(self, idx):
        img_name = self.images[idx]
        image = Image.open(os.path.join(self.image_dir, img_name)).convert('RGB')
        label = 0 if 'cat' in img_name.lower() else 1
        if self.transform:
            image = self.transform(image)
        return image, label

def get_loaders(data_dir="./datasets", batch_size=16, train_transform=None, seed=42):
    """统一划分：80% 训练 / 20% 验证，固定随机种子保证各实验可比"""
    if train_transform is None:
        train_transform = transforms.Compose([
            transforms.Resize((224, 224)),
            transforms.ToTensor(),
            transforms.Normalize(IMAGENET_MEAN, IMAGENET_STD),
        ])
    val_transform = transforms.Compose([
        transforms.Resize((224, 224)),
        transforms.ToTensor(),
        transforms.Normalize(IMAGENET_MEAN, IMAGENET_STD),
    ])
    dataset = FileNameDataset(image_dir=data_dir)
    train_size = int(0.8 * len(dataset))
    val_size = len(dataset) - train_size

    # 分别给训练/验证套上不同的 transform（验证集不做增强）
    train_set = FileNameDataset(image_dir=data_dir, transform=train_transform)
    val_set   = FileNameDataset(image_dir=data_dir, transform=val_transform)
    g = torch.Generator().manual_seed(seed)
    idx_train, idx_val = random_split(range(len(dataset)), [train_size, val_size], generator=g)

    train_loader = DataLoader(torch.utils.data.Subset(train_set, idx_train.indices),
                              batch_size=batch_size, shuffle=True)
    val_loader   = DataLoader(torch.utils.data.Subset(val_set, idx_val.indices),
                              batch_size=batch_size, shuffle=False)
    return train_loader, val_loader
```

### 0.3 公共代码（训练 + 评估函数）

也放进 `common.py`：

```python
from sklearn.metrics import accuracy_score, precision_recall_fscore_support

def train_and_eval(net, train_loader, val_loader, epochs=5, lr=1e-3, device="cpu"):
    net.to(device)
    loss_fn = torch.nn.CrossEntropyLoss()
    optimizer = torch.optim.Adam(
        [p for p in net.parameters() if p.requires_grad], lr=lr)

    for epoch in range(epochs):
        net.train()
        running_loss = 0.0
        for X, Y in train_loader:
            X, Y = X.to(device), Y.to(device)
            optimizer.zero_grad()
            loss = loss_fn(net(X), Y)
            loss.backward()
            optimizer.step()
            running_loss += loss.item() * X.size(0)
        print(f"Epoch [{epoch+1}/{epochs}] Train Loss: {running_loss/len(train_loader.dataset):.4f}")

    # 评估
    net.eval()
    all_preds, all_labels = [], []
    with torch.no_grad():
        for X, Y in val_loader:
            preds = net(X.to(device)).argmax(1).cpu()
            all_preds.extend(preds.numpy())
            all_labels.extend(Y.numpy())
    acc = accuracy_score(all_labels, all_preds)
    p, r, f1, _ = precision_recall_fscore_support(all_labels, all_preds,
                                                  average='macro', zero_division=0)
    print(f"准确率: {acc:.4f} | 精确率: {p:.4f} | 召回率: {r:.4f} | F1: {f1:.4f}")
    return acc
```

```python
class AlexNet(nn.Module):
    def __init__(self):
        """初始化"""
        super().__init__()
        self.net = nn.Sequential(
            # 这里，我们使用一个11*11的更大窗口来捕捉对象。
            # 同时，步幅为4，以减少输出的高度和宽度。
            # 另外，输出通道的数目远大于LeNet
            nn.Conv2d(3, 96, kernel_size=11, stride=4, padding=1), nn.ReLU(),
            nn.MaxPool2d(kernel_size=3, stride=2),
            # 减小卷积窗口，使用填充为2来使得输入与输出的高和宽一致，且增大输出通道数
            nn.Conv2d(96, 256, kernel_size=5, padding=2), nn.ReLU(),
            nn.MaxPool2d(kernel_size=3, stride=2),
            # 使用三个连续的卷积层和较小的卷积窗口。
            # 除了最后的卷积层，输出通道的数量进一步增加。
            # 在前两个卷积层之后，汇聚层不用于减少输入的高度和宽度
            nn.Conv2d(256, 384, kernel_size=3, padding=1), nn.ReLU(),
            nn.Conv2d(384, 384, kernel_size=3, padding=1), nn.ReLU(),
            nn.Conv2d(384, 256, kernel_size=3, padding=1), nn.ReLU(),
            nn.MaxPool2d(kernel_size=3, stride=2),
            nn.Flatten(),
            # 这里，全连接层的输出数量是LeNet中的好几倍。使用dropout层来减轻过拟合
            nn.Linear(6400, 4096), nn.ReLU(),
            nn.Dropout(p=0.5),
            nn.Linear(4096, 4096), nn.ReLU(),
            nn.Dropout(p=0.5),
            # 最后是输出层。由于这里使用Fashion-MNIST，所以用类别数为10，而非论文中的1000
            nn.Linear(4096, 2))
    
    def forward(self, X):
        """前向传播"""
        return self.net(X)
```

## 第二步：在手动模型基础上加载 ImageNet 预训练参数

### 2.1 理论讲解：迁移学习

- ImageNet 预训练模型在 **128 万张图、1000 类** 上训练过，前面的卷积层已经学会了通用的视觉特征（边缘→纹理→部件→物体），这叫**特征提取能力可以迁移**，在小样本学习场景下常常会用到迁移学习。

- 做法一（默认）：**冻结卷积层**，只训练最后的全连接层——相当于把 AlextNet 当成一个"特征提取器"，少量数据集只需要教它"这些特征怎么投票"，并不通过训练来学习不同类别的特征。

- 做法二（进阶）：**解冻全部层，用小学习率（如 1e-5）微调**，让浅层特征也向我们的任务微调，但数据少时容易过拟合，CPU 上也慢。

由于原始AlexNet为1000类的分类任务，而我们做的是2分类。因此需要把修改一层卷积层，加上一个索引方便后续更新参数。

```python
class AlexNetManual(nn.Module):
    def __init__(self, num_classes=2):
        super().__init__()
        # 卷积部分必须命名为 features，且层顺序与 torchvision 一致
        self.features = nn.Sequential(
            nn.Conv2d(3, 64, kernel_size=11, stride=4, padding=2),
            nn.ReLU(inplace=True),
            nn.MaxPool2d(kernel_size=3, stride=2),
            nn.Conv2d(64, 192, kernel_size=5, padding=2),
            nn.ReLU(inplace=True),
            nn.MaxPool2d(kernel_size=3, stride=2),
            nn.Conv2d(192, 384, kernel_size=3, padding=1),
            nn.ReLU(inplace=True),
            nn.Conv2d(384, 256, kernel_size=3, padding=1),
            nn.ReLU(inplace=True),
            nn.Conv2d(256, 256, kernel_size=3, padding=1),
            nn.ReLU(inplace=True),
            nn.MaxPool2d(kernel_size=3, stride=2),
        )
        self.avgpool = nn.AdaptiveAvgPool2d((6, 6))
        # 分类部分必须命名为 classifier，索引 6 是最后一层
        self.classifier = nn.Sequential(
            nn.Dropout(),
            nn.Linear(256 * 6 * 6, 4096),   # 索引 1
            nn.ReLU(inplace=True),
            nn.Dropout(),
            nn.Linear(4096, 4096),          # 索引 4
            nn.ReLU(inplace=True),
            nn.Linear(4096, num_classes),   # 索引 6 <- 换这里
        )

    def forward(self, x):
        x = self.features(x)
        x = self.avgpool(x)
        x = torch.flatten(x, 1)
        return self.classifier(x)

```

## 第三步：加入数据增强（三个实验横向对比）

https://zh-v2.d2l.ai/chapter_computer-vision/image-augmentation.html

### 3.1 理论讲解

数据增强 = **在不增加真实图片的前提下，人为造出"看起来不同"的训练样本**，逼模型学习本质特征而不是背图：

| 增强方式 | 模拟的现实变化 |
|---|---|
| `RandomHorizontalFlip` | 猫狗朝左还是朝右，都是猫狗 |
| `RandomResizedCrop(224, scale=(0.7,1.0))` | 拍摄距离/构图变化 |
| `ColorJitter(0.2,0.2,0.2)` | 光线、白平衡变化 |
| `RandomRotation(10)` | 相机角度轻微倾斜 |

注意：**增强只作用于训练集**，验证集必须用干净的 Resize + Normalize，否则评估指标不可信。

## 第四步：调用 `torchvision.models` 封装好的模型横向对比

### 4.1 理论讲解

掌握手动搭建后，实际工程中直接使用 `torchvision.models` 里的成熟模型。选型维度：

| 模型 | 参数量 | CPU 推理速度 | 特点 |
|---|---|---|---|
| `resnet18` | 11M | 快 | 结构规整，最经典的入门模型 |
| `mobilenet_v3_small` | 2.5M | 很快 | 深度可分离卷积，为手机端设计 |
| `shufflenet_v2_x1_0` | 2.3M | 很快 | 通道混洗，极致轻量 |
| `efficientnet_b0` | 5M | 中 | 复合缩放，精度/开销比优秀（CPU 上较慢，选做） |



### 4.2 参考代码

```python
import torch.nn as nn
from torchvision import models
from common import get_loaders, train_and_eval

def build_model(name, num_classes=2, freeze=True):
    """统一接口：加载预训练模型并替换分类头"""
    constructors = {
        "resnet18":            lambda: models.resnet18(weights=models.ResNet18_Weights.DEFAULT),
        "mobilenet_v3_small":  lambda: models.mobilenet_v3_small(weights=models.MobileNet_V3_Small_Weights.DEFAULT),
        "shufflenet_v2_x1_0":  lambda: models.shufflenet_v2_x1_0(weights=models.ShuffleNet_V2_X1_0_Weights.DEFAULT),
    }
    net = constructors[name]()

    # 找到并替换分类头（不同模型的头名字不一样）
    if name == "resnet18":
        net.fc = nn.Linear(net.fc.in_features, num_classes)
        head_params = net.fc.parameters()
    elif name == "mobilenet_v3_small":
        net.classifier[3] = nn.Linear(net.classifier[3].in_features, num_classes)
        head_params = net.classifier[3].parameters()
    elif name == "shufflenet_v2_x1_0":
        net.fc = nn.Linear(net.fc.in_features, num_classes)
        head_params = net.fc.parameters()

    if freeze:  # 冻结特征提取层，只训练分类头（CPU 友好）
        for p in net.parameters():
            p.requires_grad = False
        for p in head_params:
            p.requires_grad = True
    return net

if __name__ == '__main__':
    train_loader, val_loader = get_loaders("./datasets", batch_size=16)
    results = {}
    for name in ["resnet18", "mobilenet_v3_small", "shufflenet_v2_x1_0"]:
        print(f"\n===== 模型: {name} =====")
        net = build_model(name)
        acc = train_and_eval(net, train_loader, val_loader, epochs=5, lr=1e-3)
        results[name] = acc
        torch.save(net.state_dict(), f"model_{name}.pth")   # 顺手保存，第五步要用

    print("\n===== 横向对比结果 =====")
    for name, acc in sorted(results.items(), key=lambda x: -x[1]):
        print(f"{name:24s} 准确率: {acc:.4f}")
```

## 第五步：模型保存与单张图片测试

### 5.1 理论讲解

- **只保存参数**（推荐）：`torch.save(net.state_dict(), "best_model.pth")`，文件小、跨版本稳定；加载时需要先构建同结构模型再 `load_state_dict`。
- **保存整个模型**：`torch.save(net, "model.pt")`，方便但会把代码路径也序列化，容易踩坑，不推荐。
- 工程惯例：训练过程中**记录验证集最佳准确率，只保存 best 模型**（早停思想）。

### 5.2 训练时保存最佳模型

```python
import torch
import torch.nn as nn
from torchvision import models
from common import get_loaders, IMAGENET_MEAN, IMAGENET_STD

def train_with_best_save(net, train_loader, val_loader, epochs=5, lr=1e-3, save_path="best_model.pth"):
    device = "cpu"
    net.to(device)
    loss_fn = nn.CrossEntropyLoss()
    optimizer = torch.optim.Adam([p for p in net.parameters() if p.requires_grad], lr=lr)

    best_acc = 0.0
    for epoch in range(epochs):
        net.train()
        for X, Y in train_loader:
            optimizer.zero_grad()
            loss = loss_fn(net(X), Y)
            loss.backward()
            optimizer.step()

        # 验证
        net.eval()
        correct = total = 0
        with torch.no_grad():
            for X, Y in val_loader:
                correct += (net(X).argmax(1) == Y).sum().item()
                total += Y.size(0)
        acc = correct / total
        print(f"Epoch [{epoch+1}/{epochs}] Val Acc: {acc:.4f}")

        if acc > best_acc:                      # 只保留历史最佳
            best_acc = acc
            torch.save(net.state_dict(), save_path)
            print(f"  ^ 最佳模型已保存到 {save_path}")
    print(f"最佳准确率: {best_acc:.4f}")

if __name__ == '__main__':
    net = models.resnet18(weights=models.ResNet18_Weights.DEFAULT)
    net.fc = nn.Linear(net.fc.in_features, 2)
    for p in net.parameters():
        p.requires_grad = False
    for p in net.fc.parameters():
        p.requires_grad = True
    train_loader, val_loader = get_loaders("./datasets", batch_size=16)
    train_with_best_save(net, train_loader, val_loader, epochs=5, lr=1e-3)
```

### 5.3 单张图片推理测试，参考代码

```python
import sys
from PIL import Image
import torch
import torch.nn as nn
from torchvision import models, transforms

CLASSES = ["cat 猫", "dog 狗"]

def load_model(weight_path="best_model.pth"):
    net = models.resnet18(weights=None)   # 这里使用的是resnet18, 结构一致即可，不用下载权重
    net.fc = nn.Linear(net.fc.in_features, 2)
    net.load_state_dict(torch.load(weight_path, map_location="cpu"))
    net.eval()
    return net

def predict(image_path, net):
    transform = transforms.Compose([
        transforms.Resize((224, 224)),
        transforms.ToTensor(),
        transforms.Normalize([0.485, 0.456, 0.406], [0.229, 0.224, 0.225]),
    ])
    img = Image.open(image_path).convert("RGB")
    x = transform(img).unsqueeze(0)              # 增加 batch 维 -> [1,3,224,224]
    with torch.no_grad():
        probs = torch.softmax(net(x), dim=1)[0]  # 得分转概率
    pred = probs.argmax().item()
    return CLASSES[pred], probs[pred].item(), probs

if __name__ == '__main__':
    net = load_model(sys.argv[1] if len(sys.argv) > 1 else "best_model.pth")
    path = sys.argv[2] if len(sys.argv) > 2 else "./datasets/cat.1.jpg"
    label, conf, probs = predict(path, net)
    print(f"图片: {path}")
    print(f"预测: {label}，置信度 {conf:.2%}")
    print(f"各类概率: 猫 {probs[0]:.2%} | 狗 {probs[1]:.2%}")
```

## 第六步：封装成可拍照分类的小程序（Gradio，5 分钟上线）

### 6.1 理论讲解：从 `.py` 到"应用"

部署三要素：**加载训练好的权重 → 接收用户输入（拍照/上传）→ 返回预测结果**。

我们用 `gradio`（`pip install gradio`）——一个专门给模型套界面的库，自带**摄像头拍照**输入组件。启动后：
- 本机浏览器访问 `http://127.0.0.1:7860`；
- 手机连同一 Wi-Fi，访问 `http://<老师电脑IP>:7860`，即可用手机摄像头拍照测试——这就是课堂上的"小程序"。
- （延伸）若要做成真正的**微信小程序**，只需把 predict 逻辑换成 FastAPI 提供 `/predict` 接口，小程序端用 `wx.uploadFile` 调用即可，思路完全一致。

### 6.2 参考代码

```python
"""拍照猫狗分类小程序：python step6_app.py 后按提示访问"""
import torch
import torch.nn as nn
import gradio as gr
from PIL import Image
from torchvision import models, transforms

CLASSES = {0: "🐱 猫", 1: "🐶 狗"}

# ---------- 1. 加载模型（启动时只做一次） ----------
net = models.resnet18(weights=None)
net.fc = nn.Linear(net.fc.in_features, 2)
net.load_state_dict(torch.load("best_model.pth", map_location="cpu"))
net.eval()

transform = transforms.Compose([
    transforms.Resize((224, 224)),
    transforms.ToTensor(),
    transforms.Normalize([0.485, 0.456, 0.406], [0.229, 0.224, 0.225]),
])

# ---------- 2. 定义预测函数（gradio 每次拍照/上传都会调用） ----------
def predict(img: Image.Image):
    img = img.convert("RGB")
    x = transform(img).unsqueeze(0)
    with torch.no_grad():
        probs = torch.softmax(net(x), dim=1)[0]
    conf, pred = probs.max(0)
    detail = {CLASSES[i]: f"{p:.2%}" for i, p in enumerate(probs)}
    return f"识别结果：{CLASSES[int(pred)]}（置信度 {conf:.2%}）", detail

# ---------- 3. 搭建界面并启动 ----------
demo = gr.Interface(
    fn=predict,
    inputs=gr.Image(type="pil", sources=["webcam", "upload"], label="拍照或上传"),
    outputs=[gr.Textbox(label="识别结果"), gr.Label(label="各类别概率")],
    title="🐱🐶 猫狗分类器",
    description="基于 ResNet18 迁移学习训练。对准猫或狗拍照，或上传图片即可识别。",
)

if __name__ == "__main__":
    demo.launch(server_name="0.0.0.0", server_port=7860)
    # 本机访问: http://127.0.0.1:7860
    # 手机(同一局域网)访问: http://<电脑IP>:7860  例如 http://192.168.1.100:7860
```

## 总结：六步实验的完整对比

| 步骤 | 核心改动 | 预期准确率 | 核心结论 |
|---|---|---|---|
| 1. 手动搭建+从零训练 | 随机权重 | 0.65~0.80 | 小数据集从零训练学不出通用视觉特征 |
| 2. +ImageNet 预训练 | 灌入预训练权重、冻结卷积 | 0.90~1.00 | 迁移学习是小数据集的最优解 |
| 3. +数据增强 | 训练集随机翻转/裁剪/色彩 | ≈步骤2 或更稳 | 增强提升泛化，数据越多越明显 |
| 4. 多模型对比 | resnet18 / mobilenet / shufflenet | 都在 0.9+ | 小数据下差距小，选型看速度与体积 |
| 5. 保存与推理 | state_dict 保存 + 单图测试 | — | 训练与推理分离是工程标配 |
| 6. 部署小程序 | Gradio 摄像头界面 | — | 模型只有被用起来才有价值 |
